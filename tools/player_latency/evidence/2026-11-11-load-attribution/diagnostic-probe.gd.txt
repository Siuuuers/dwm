extends "res://tests/integration/compact_history_restore_child.gd"

var measured_loads := 0
var _ready_at := 0
var _samples: Array[Dictionary] = []

func _dispatch_load(control: Control) -> Dictionary:
	assert_true(is_instance_valid(control) and control.is_visible_in_tree() and not control.disabled)
	var overlay := get_tree().root.find_child("GutLayer", true, false)
	if overlay is CanvasLayer: overlay.hide()
	var point := control.get_global_transform_with_canvas() * (control.size / 2.0)
	var events: Array[InputEvent] = []
	for pressed: bool in [true, false]:
		var motion := InputEventMouseMotion.new()
		motion.position = point
		motion.global_position = point
		motion.button_mask = 0 if pressed else MOUSE_BUTTON_MASK_LEFT
		events.append(motion)
		var event := InputEventMouseButton.new()
		event.position = point
		event.global_position = point
		event.button_index = MOUSE_BUTTON_LEFT
		event.button_mask = MOUSE_BUTTON_MASK_LEFT if pressed else 0
		event.pressed = pressed
		events.append(event)
	var started := Time.get_ticks_usec()
	for event: InputEvent in events: get_viewport().push_input(event, true)
	return {"started_us": started, "dispatch_us": Time.get_ticks_usec() - started}


func _load_authored(slot: int, proof: Dictionary) -> bool:
	assert_eq(ProjectSettings.globalize_path("user://"), proof.user_dir)
	assert_eq(AUTHORED_MANIFEST.scene_registration_fingerprint(), proof.registration_sha256)
	var path := "user://saves/slot_%d.json" % slot
	assert_eq(FileAccess.get_sha256(path), proof.slot_sha256)
	var profile_before: Dictionary = profile.get_profile_snapshot().duplicate(true)
	var app: Control
	if measured_loads == 0:
		get_node("/root/SceneRouter").goto_menu()
		if not await _wait_route(MENU): return false
		var menu: Node = get_tree().current_scene
		saves.live_session_ready.connect(func() -> void: _ready_at = Time.get_ticks_usec())
		if not await _press_authored(menu.get_node("%LogInButton")): return false
		app = menu._backup_app_instance
	else:
		var pause: Node = get_node("/root/SceneRouter")._production_pause
		var opened: Dictionary = await pause.open_scene_backup("load")
		assert_true(opened.get("ok", false))
		if not opened.get("ok", false): return false
		app = pause.surface._hosts[&"backup"]
	ready_count = 0
	assert_not_null(app, "Title LogIn mounts Backup")
	if not is_instance_valid(app): return false
	if not await _press_authored(app.drawer_buttons.get("slot:%d" % slot)): return false
	var expanded: Dictionary = preload("res://scripts/narrative/SoloReadingSession.gd").expand_player_checkpoint(proof.checkpoint)
	if not _ok(expanded): return false
	var line_id: String = expanded.value.reading_session.frontier.line_id
	_ready_at = 0
	var action := _dispatch_load(app.action_buttons.get("load"))
	var confirmation := {}
	if is_instance_valid(app) and is_instance_valid(app.confirmation):
		await get_tree().process_frame
		confirmation = _dispatch_load(app.confirmation.confirm_button)
	var committed: Dictionary = action if confirmation.is_empty() else confirmation
	if not await _await_ready(): return false
	var deadline := Time.get_ticks_msec() + 30000
	var usable_at := 0
	while Time.get_ticks_msec() < deadline:
		var caption: Node = _caption()
		if bridge._runtime_adapter.current_line_id() == line_id \
				and bridge._runtime_adapter.is_current_line_complete() \
				and bridge.is_current_line_presentation_acknowledged() \
				and is_instance_valid(caption) and caption.caption_text.is_visible_in_tree() \
				and not caption.caption_text.revealing \
				and not bridge._runtime_adapter._dialogic.get_subsystem("Inputs").is_input_blocked():
			usable_at = Time.get_ticks_usec()
			break
		await get_tree().process_frame
	var post_draw_at: Variant = null
	if DisplayServer.get_name() != "headless":
		await RenderingServer.frame_post_draw
		post_draw_at = Time.get_ticks_usec()
	assert_gt(_ready_at, int(committed.started_us))
	assert_gt(usable_at, 0)
	if usable_at == 0: return false
	_samples.append({"index": measured_loads, "first_process_load": measured_loads == 0,
		"confirmation_required": not confirmation.is_empty(), "action_dispatch_us": action.dispatch_us,
		"commit_dispatch_us": committed.dispatch_us, "commit_to_ready_us": _ready_at - int(committed.started_us),
		"commit_to_usable_us": usable_at - int(committed.started_us),
		"action_to_usable_us": usable_at - int(action.started_us),
		"commit_to_post_draw_us": null if post_draw_at == null else post_draw_at - int(committed.started_us)})
	print("CHECKPOINT_LOAD_SAMPLE " + JSON.stringify(_samples.back()))
	measured_loads += 1
	_ready_observation = _native_observation()
	if not await _wait_line(line_id): return false
	assert_eq(_ready_observation.line_id, line_id)
	assert_true(_ready_observation.visible)
	assert_false(_ready_observation.revealing)
	assert_true(_ready_observation.event_done)
	assert_eq(float(_ready_observation.visible_ratio), 1.0)
	assert_eq(_ready_observation.ledger, expanded.value.reading_session.ledger)
	assert_eq(bridge.capture_reading_checkpoint(false).value, expanded.value)
	var current: Dictionary = game.capture_run_snapshot_input()
	assert_eq(current.lifecycle.run_id, proof.snapshot.lifecycle.run_id)
	assert_eq(current.lifecycle.branch_id, proof.snapshot.lifecycle.branch_id)
	for key: String in ["gameplay", "scene"]:
		assert_eq(current[key], proof.snapshot[key], "Load preserves " + key)
	assert_eq(profile.get_profile_snapshot(), profile_before, "Load preserves current profile")
	assert_eq(FileAccess.get_sha256(path), proof.slot_sha256)
	await _screenshot("authored-restored-slot-%d" % slot)
	return true


func test_checkpoint_load_cost() -> void:
	await load_older_save_preserves_later_global_discoveries()
	var proof: Dictionary = PARSER.parse_object(FileAccess.get_file_as_string("user://compact-history-proof.json")).value
	for attempt in range(3):
		if not await _load_authored(1, proof): return
	var report := {"revision": OS.get_environment("DWM_BENCHMARK_REVISION"),
		"engine": Engine.get_version_info().string, "display": DisplayServer.get_name(), "samples": _samples,
		"limits": "Fresh process after services and expected-content setup, shared OS cache. Native viewport dispatch has no helper frame waits. Repeated Load requires confirmation; one layout frame precedes its input. Session-ready signal records only a scalar; assertions follow usability timing. Headless readiness is not physical display latency."}
	print("CHECKPOINT_LOAD_REPORT " + JSON.stringify(report))
	_write_json("user://checkpoint-load-latency.json", report)
