import os,subprocess,pathlib,json,shutil
root=pathlib.Path.cwd();out=root/'.godot/ci/first-caption';user=out/'matched-user';rows=[]
for i,label in enumerate(['baseline','candidate','candidate','baseline']):
 if user.exists():shutil.rmtree(user)
 env=os.environ.copy();env.pop('DWM_TEST_ROOT',None);env.update(DWM_STARTUP_AUDIT='1',XDG_DATA_HOME=str(user),DWM_EARLY_COMPUTER='normal')
 repo=root.parent/'dwm-caption-baseline' if label=='baseline' else root
 log=out/f'final-{i}-{label}.log'

 s=log.read_text();assert p.returncode==0 and 'ERROR:' not in s,s

 rows.append(row);(out/'final-matched.json').write_text(json.dumps(rows,indent=2))
